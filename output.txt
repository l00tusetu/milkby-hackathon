```
    const langs: Language[] = ['ru', 'be', 'en', 'zh'];
    return langs.map((lang) => {
      let total = 0;
      let filled = 0;
      for (const p of products) {
        total += 2;
        if (p.name[lang]?.trim()) filled++;
        if (p.description[lang]?.trim()) filled++;
      }
      for (const n of news) {
        total += 2;
        if (n.title[lang]?.trim()) filled++;
        if (n.excerpt[lang]?.trim()) filled++;
      }
      for (const key of Object.keys(translations)) {
        total += 1;
        if (translations[key]?.[lang]?.trim()) filled++;
      }
      const pct = total > 0 ? Math.round((filled / total) * 100) : 100;
      return { lang, filled, total, pct };
```

```
import { useCatalogStore } from '../hooks/useCatalogStore';

import { SearchBar } from '../components/SearchBar';
import { Filters, CatalogFilterState } from '../components/Filters';
import { ProductGrid } from '../components/ProductGrid';
import { SmartFinder } from '../components/SmartFinder';

interface CatalogProps {
  initialCategorySlug?: string;
}

const DEFAULT_FILTERS: CatalogFilterState = {
  categoryId: 'all',
  subcategoryId: 'all',
  fatRange: 'all',
  lactoseFreeOnly: false,
  highProteinOnly: false,
  exportReadyOnly: false,
  sortBy: 'featured',
};

export const Catalog: React.FC<CatalogProps> = ({ initialCategorySlug }) => {
  const { language, localize, t } = useLanguage();
  const { products, categories } = useCatalogStore();
  const [searchParams] = useSearchParams();

  const [query, setQuery] = useState(() => searchParams.get('q') || '');
  const [filterState, setFilterState] = useState<CatalogFilterState>(() => ({
    ...DEFAULT_FILTERS,
    categoryId: initialCategorySlug || 'all',
  }));

  useEffect(() => {
    setFilterState((prev) => ({
      ...prev,
      categoryId: initialCategorySlug || 'all',
      subcategoryId: 'all',
    }));
  }, [initialCategorySlug]);

  const activeCategory = categories.find((c) => c.id === filterState.categoryId);

  const filteredProducts = useMemo(() => {
    const q = query.trim().toLowerCase();

    const list = products
      .filter((p) => p.visible)
      .filter((p) => {
        if (filterState.categoryId !== 'all' && p.category !== filterState.categoryId) {
          return false;
        }
        if (filterState.subcategoryId !== 'all' && p.subcategory !== filterState.subcategoryId) {
          return false;
        }
        if (filterState.fatRange !== 'all') {
          const fat = p.fatContent ?? 0;
          if (filterState.fatRange === 'low' && fat > 5) return false;
          if (filterState.fatRange === 'medium' && (fat < 15 || fat > 35)) return false;
          if (filterState.fatRange === 'high' && fat < 45) return false;
        }
        if (filterState.lactoseFreeOnly && !p.lactoseFree) return false;
        if (filterState.highProteinOnly && (p.proteinContent ?? 0) < 8) return false;
        if (filterState.exportReadyOnly && !p.exportReady) return false;

        if (q) {
          const cat = categories.find((c) => c.id === p.category);
          const haystack = [
            p.name.ru,
            p.name.be,
            p.name.en,
            p.name.zh,
            p.description.ru,
            p.description.en,
            cat?.name.ru || '',
            cat?.name.be || '',
            cat?.name.en || '',
            cat?.name.zh || '',
            p.useCases.join(' '),
          ]
            .join(' ')
            .toLowerCase();
          if (!haystack.includes(q)) return false;
        }
        return true;
      });

    const sorted = [...list];
    sorted.sort((a, b) => {
      if (filterState.sortBy === 'fat-asc') return (a.fatContent ?? 0) - (b.fatContent ?? 0);
      if (filterState.sortBy === 'fat-desc') return (b.fatContent ?? 0) - (a.fatContent ?? 0);
      if (filterState.sortBy === 'protein-desc')
        return (b.proteinContent ?? 0) - (a.proteinContent ?? 0);
      if (filterState.sortBy === 'name')
        return localize(a.name).localeCompare(localize(b.name), language);
      return Number(Boolean(b.featured)) - Number(Boolean(a.featured));
    });

    return sorted;
  }, [products, categories, filterState, query, localize, language]);

  const handleReset = () => {
    setQuery('');
    setFilterState({

      categoryId: initialCategorySlug || 'all',

  };

    : t('navCatalog');

    ? localize(activeCategory.description)
    : language === 'ru'

    : language === 'be'

    : language === 'zh'

    : 'Complete dairy catalog with dynamic filtering by category, fat content, protein, lactose-free status, and packaging format.';

  const seoTitle = activeCategory

    : `${t('navCatalog')} — ${t('brandName')} (milk.by)`;

  const seoDesc = activeCategory
    ? localize(activeCategory.seo.description)
    : pageDesc;

  return (

      <SeoHead
        title={seoTitle}
        description={seoDesc}

      />

      {/* Breadcrumbs & Page Heading */}
      <div className="space-y-3">
        <nav className="text-xs text-slate-500 flex items-center gap-2" aria-label="Breadcrumb">
          <Link to="/" className="hover:text-[#0057B8]">
            {language === 'ru' ? 'Главная' : language === 'be' ? 'Галоўная' : language === 'zh' ? '首页' : 'Home'}
```

```
    const uniqueTitles = new Set(titles);
    return {
      totalProducts: visibleProducts.length,
      uniqueTitlesCount: uniqueTitles.size,
      noDuplicates: titles.length === uniqueTitles.size,
    };
  }, [products, language]);

    const blob = new Blob([xmlContent], { type: 'application/xml;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'sitemap.xml';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  };

    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8 sm:py-10 space-y-10">
```

```
import React, { useEffect } from 'react';
import { useLanguage } from '../hooks/useLanguage';

interface SeoHeadProps {
  title: string;
  description: string;
  canonicalHash?: string;
}

export const SeoHead: React.FC<SeoHeadProps> = ({ title, description, canonicalHash }) => {
  const { language } = useLanguage();

  useEffect(() => {
    document.title = title;
    document.documentElement.lang = language;

    const setMeta = (selector: string, attr: string, value: string) => {
      let el = document.querySelector(selector);
      if (!el) {
        el = document.createElement('meta');
        if (selector.includes('property=')) {
          el.setAttribute('property', selector.match(/property="([^"]+)"/)?.[1] || '');
        } else {
          el.setAttribute('name', selector.match(/name="([^"]+)"/)?.[1] || '');
        }
        document.head.appendChild(el);
      }
      el.setAttribute(attr, value);
    };

    setMeta('meta[name="description"]', 'content', description);
    setMeta('meta[property="og:title"]', 'content', title);
    setMeta('meta[property="og:description"]', 'content', description);

    if (canonicalHash !== undefined) {
      let link = document.querySelector('link[rel="canonical"]');
      if (!link) {
        link = document.createElement('link');
        link.setAttribute('rel', 'canonical');
        document.head.appendChild(link);
      }
      link.setAttribute('href', `https://milk.by/#${canonicalHash}`);
    }
  }, [title, description, canonicalHash, language]);

  return null;
};
```

```
import { useLanguage } from '../hooks/useLanguage';
import { useCatalogStore } from '../hooks/useCatalogStore';
import { runSmartProductMatch } from '../utils/smartMatch';
import { ImageWithFallback } from './ImageWithFallback';

const SAMPLE_PROMPTS = {
  ru: [
    'Мне нужен сыр для пиццы',
    'Мне нужен продукт без лактозы',
    'Мне нужен продукт с высоким содержанием белка',
    'Мне нужен сыр для дегустации с вином',
  ],
  be: [
    'Мне патрэбен сыр для піцы',
    'Мне патрэбен прадукт без лактозы',
    'Мне патрэбен прадукт з высокім утрыманнем бялку',
    'Мне патрэбен сыр для дэгустацыі з віном',
  ],
  en: [
    'I need cheese for pizza',
    'I need a lactose-free product',
    'I need a high-protein dairy product',
    'I need aged cheese for wine tasting',
  ],
  zh: [
    '我需要披萨专用奶酪',
    '我需要无乳糖乳制品',
    '我需要高蛋白乳制品',
    '我需要配葡萄酒的奶酪',
  ],
};

export const SmartFinder: React.FC = () => {
  const { t, language, localize } = useLanguage();
  const { products } = useCatalogStore();
  const [query, setQuery] = useState('');

  const matches = useMemo(
    () => runSmartProductMatch(query, products, language),
    [query, products, language]
  );

  const prompts = SAMPLE_PROMPTS[language] || SAMPLE_PROMPTS.ru;

  return (
    <section className="border border-slate-200 rounded-2xl bg-slate-50/70 p-6 sm:p-8 space-y-5">
      <div className="flex flex-col md:flex-row md:items-end justify-between gap-4">
        <div className="space-y-1.5">
          <div className="inline-flex items-center gap-2 text-xs font-semibold text-[#0057B8]">
            <Compass className="w-4 h-4" />
            <span>{t('smartFinderTitle')}</span>
          </div>
          <p className="text-sm text-slate-600 max-w-2xl">
            {t('smartFinderSubtitle')}
          </p>
        </div>
      </div>

      <div className="flex flex-col sm:flex-row gap-3">
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder={t('smartFinderPlaceholder')}
          className="flex-1 h-11 px-4 text-sm bg-white border border-slate-200 rounded-lg text-slate-900 placeholder:text-slate-400 focus:outline-none focus:border-[#0057B8] focus:ring-2 focus:ring-[#0057B8]/15"
        />
        {query && (
          <button
            type="button"
            onClick={() => setQuery('')}
            className="h-11 px-4 text-xs font-medium text-slate-700 bg-white border border-slate-200 rounded-lg hover:bg-slate-100 transition-colors whitespace-nowrap shrink-0"
          >
            {language === 'ru' ? 'Очистить' : language === 'be' ? 'Ачысціць' : language === 'zh' ? '清空' : 'Clear'}
          </button>
        )}
      </div>

      <div className="flex flex-wrap gap-2">
        {prompts.map((sample) => (
          <button
            key={sample}
            type="button"
            onClick={() => setQuery(sample)}
            className={`px-3 py-1.5 text-xs font-medium rounded-lg border transition-colors whitespace-nowrap ${
              query === sample
                ? 'bg-[#0057B8] text-white border-[#0057B8]'
                : 'bg-white text-slate-700 border-slate-200 hover:border-slate-300 hover:text-slate-900'
            }`}
          >
            {sample}
          </button>
        ))}
      </div>

      {query.trim().length > 0 && (
        <div className="pt-2">
          {matches.length > 0 ? (
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
              {matches.map(({ product, reasons }) => (
                <Link
                  key={product.id}
                  to={`/product/${product.slug}`}
                  className="group bg-white border border-slate-200 rounded-xl p-4 flex gap-4 hover:border-[#0057B8]/50 hover:shadow-sm transition-all"
                >
                  <div className="w-20 h-20 rounded-lg bg-slate-50 overflow-hidden shrink-0 border border-slate-100">
                    <ImageWithFallback
                      src={product.image}
                      alt={localize(product.name)}
                      className="w-full h-full object-cover"
                    />
                  </div>
                  <div className="min-w-0 flex-1 space-y-1.5">
                    <div className="text-sm font-semibold text-slate-900 group-hover:text-[#0057B8] transition-colors truncate">
                      {localize(product.name)}
                    </div>
                    <ul className="space-y-0.5">
                      {reasons.slice(0, 2).map((reason, idx) => (
                        <li
                          key={idx}
                          className="text-xs text-slate-600 flex items-start gap-1.5 leading-snug"
                        >
                          <Check className="w-3.5 h-3.5 text-[#0057B8] shrink-0 mt-0.5" />
                          <span>{reason}</span>
                        </li>
                      ))}
                    </ul>
                    <div className="pt-1 text-xs font-medium text-[#0057B8] inline-flex items-center gap-1">
                      <span>{product.packages.map((p) => p.weight).slice(0, 2).join(' · ')}</span>
                      <ArrowRight className="w-3 h-3" />
                    </div>
                  </div>
                </Link>
              ))}
            </div>
          ) : (
            <p className="text-sm text-slate-600 bg-white border border-slate-200 rounded-lg p-4">
              {language === 'ru'
                ? 'Подходящих позиций по этому запросу не найдено. Попробуйте «сыр для пиццы», «без лактозы» или «высокий белок».'
                : language === 'be'
                ? 'Падыходзячых пазіцый не знойдзена.'
                : language === 'zh'
                ? '未找到匹配产品，请尝试“披萨奶酪”、“无乳糖”或“高蛋白”。'
                : 'No exact match found. Try "pizza cheese", "lactose-free", or "high protein".'}
            </p>
          )}
        </div>
      )}
    </section>
  );
};
```
